package pwp

import (
	"context"
	"net"
)


type PeerConn interface {
    SendMessage(msg Message) error
    ReadMessage() (Message, error)
    Close() error
    RemoteAddr() string
}

// Handshake separate from regular messages
func Handshake(conn net.Conn, infoHash [20]byte, peerID [20]byte) ([20]byte, error)

// ============================================================================
// 3. PEER - Stateful peer abstraction
// ============================================================================

type Peer struct {
    ID       [20]byte
    Conn     PeerConn
    Bitfield *Bitfield
    
    // State
    AmChoking      bool
    AmInterested   bool
    PeerChoking    bool
    PeerInterested bool
    
    // Channels for communication with downloader
    incoming chan Message      // Messages received from this peer
    outgoing chan Message      // Messages to send to this peer
    close    chan struct{}     // Signal to stop peer goroutines
}

func NewPeer(conn PeerConn, peerID [20]byte, bitfield *Bitfield) *Peer

// Start peer's read/write loops
func (p *Peer) Start()

// Stop peer's goroutines
func (p *Peer) Stop()

// Check if peer has piece
func (p *Peer) HasPiece(index uint32) bool

// use SerializeMessage(m Message) 
// High-level message sending
func (p *Peer) SendChoke() error
func (p *Peer) SendUnchoke() error
func (p *Peer) SendInterested() error
func (p *Peer) SendNotInterested() error
func (p *Peer) SendHave(pieceIndex uint32) error
func (p *Peer) SendRequest(index, begin, length uint32) error
func (p *Peer) SendPiece(index, begin uint32, data []byte) error
func (p *Peer) SendCancel(index, begin, length uint32) error

// ============================================================================
// 4. BITFIELD - Track which pieces peer has
// ============================================================================

type Bitfield struct {
    bits []byte
}

func NewBitfield(numPieces int) *Bitfield
func (b *Bitfield) HasPiece(index uint32) bool
func (b *Bitfield) SetPiece(index uint32)
func (b *Bitfield) ClearPiece(index uint32)
func (b *Bitfield) NumPieces() int
func (b *Bitfield) MissingPieces() []uint32

// ============================================================================
// 5. WORK QUEUE - Thread-safe piece queue
// ============================================================================

type PieceWork struct {
    Index  uint32
    Hash   [20]byte
    Length uint32
}

type WorkQueue struct {
    queue chan PieceWork
}

func NewWorkQueue(pieces []PieceWork) *WorkQueue
func (wq *WorkQueue) Pop() (PieceWork, bool)
func (wq *WorkQueue) Push(work PieceWork)
func (wq *WorkQueue) Len() int

// ============================================================================
// 6. DOWNLOADER - Main coordinator
// ============================================================================

type DownloadResult struct {
    Index uint32
    Data  []byte
    Err   error
}

type Downloader struct {
    peers     []*Peer
    workQueue *WorkQueue
    results   chan DownloadResult  // Results channel for user
    done      chan struct{}        // Internal signal for completion
    blockSize uint32
}

func NewDownloader(peers []*Peer, pieces []PieceWork, blockSize uint32) *Downloader

// Start downloading, returns channel for results
func (d *Downloader) Start(ctx context.Context) <-chan DownloadResult

// Stop all downloads
func (d *Downloader) Stop()

// Internal worker that runs per peer
func (d *Downloader) worker(ctx context.Context, peer *Peer)

// Download single piece from peer (breaks into blocks)
func (d *Downloader) downloadPiece(ctx context.Context, peer *Peer, work PieceWork) ([]byte, error)

// Request single block from peer
func (d *Downloader) requestBlock(ctx context.Context, peer *Peer, index, begin, length uint32) ([]byte, error)

// ============================================================================
// 7. TORRENT METADATA - Info about file/pieces
// ============================================================================

type TorrentInfo struct {
    PieceLength uint32
    Pieces      [][20]byte  // SHA1 hashes
    TotalLength uint64
    InfoHash    [20]byte
}

func (t *TorrentInfo) NumPieces() int
func (t *TorrentInfo) GetPieceLength(index uint32) uint32

// ============================================================================
// 8. HELPER FUNCTIONS
// ============================================================================

// Verify piece hash
func VerifyPiece(data []byte, hash [20]byte) bool

// Parse .torrent file
func ParseTorrentFile(path string) (*TorrentInfo, error)
